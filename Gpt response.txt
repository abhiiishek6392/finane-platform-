Recommended Database Schema
User
model User {
  id            String   @id @default(cuid())
  clerkUserId   String   @unique
  email         String   @unique
  name          String?
  accounts      Account[]
  budgets       Budget[]
  reports       AIReport[]
  notifications Notification[]
  createdAt     DateTime @default(now())
}
Account
model Account {
  id           String        @id @default(cuid())
  name         String
  type         String
  balance      Float
  userId       String
  user         User          @relation(fields: [userId], references: [id])
  transactions Transaction[]
  createdAt    DateTime @default(now())
}
Transaction
model Transaction {
  id            String   @id @default(cuid())
  amount        Float
  type          String
  category      String
  description   String?
  date          DateTime
  recurring     Boolean  @default(false)
  receiptUrl    String?
  accountId     String
  account       Account  @relation(fields: [accountId], references: [id])
  createdAt     DateTime @default(now())
}
Budget
model Budget {
  id          String   @id @default(cuid())
  limit       Float
  spent       Float    @default(0)
  month       Int
  year        Int
  userId      String
  user        User     @relation(fields: [userId], references: [id])
  createdAt   DateTime @default(now())
}
